CREATE SCHEMA IF NOT EXISTS nosql;

-- Таблица UserSessions
CREATE TABLE nosql.UserSessions (
    id SERIAL PRIMARY KEY,
    json_data JSONB NOT NULL,
    session_id TEXT,
    user_id TEXT,
    start_time TIMESTAMP,
    end_time TIMESTAMP,
    pages_visited JSONB,
    device JSONB,
    actions JSONB
);

-- Таблица ProductPriceHistory
CREATE TABLE nosql.ProductPriceHistory (
    id SERIAL PRIMARY KEY,
    json_data JSONB NOT NULL,
    product_id TEXT,
    price_changes JSONB,
    current_price NUMERIC,
    currency TEXT
);

-- Таблица EventLogs
CREATE TABLE nosql.EventLogs (
    id SERIAL PRIMARY KEY,
    json_data JSONB NOT NULL,
    event_id TEXT,
    timestamp TIMESTAMP,
    event_type TEXT,
    details JSONB
);

-- Таблица SupportTickets
CREATE TABLE nosql.SupportTickets (
    id SERIAL PRIMARY KEY,
    json_data JSONB NOT NULL,
    ticket_id TEXT,
    user_id TEXT,
    status TEXT,
    issue_type TEXT,
    messages JSONB,
    created_at TIMESTAMP,
    updated_at TIMESTAMP
);

-- Таблица UserRecommendations
CREATE TABLE nosql.UserRecommendations (
    id SERIAL PRIMARY KEY,
    json_data JSONB NOT NULL,
    user_id TEXT,
    recommended_products JSONB,
    last_updated TIMESTAMP
);

-- Таблица ModerationQueue
CREATE TABLE nosql.ModerationQueue (
    id SERIAL PRIMARY KEY,
    json_data JSONB NOT NULL,
    review_id TEXT,
    user_id TEXT,
    product_id TEXT,
    review_text TEXT,
    rating INTEGER,
    moderation_status TEXT,
    flags JSONB,
    submitted_at TIMESTAMP
);

-- Таблица SearchQueries
CREATE TABLE nosql.SearchQueries (
    id SERIAL PRIMARY KEY,
    json_data JSONB NOT NULL,
    query_id TEXT,
    user_id TEXT,
    query_text TEXT,
    timestamp TIMESTAMP,
    filters JSONB,
    results_count INTEGER
);

--Витрина: Статистика пользовательских сессий
DROP MATERIALIZED VIEW nosql.session_analytics;
CREATE MATERIALIZED VIEW nosql.session_analytics AS
SELECT 
    user_id,
    COUNT(*) AS total_sessions,
    AVG(EXTRACT(EPOCH FROM (end_time - start_time)))::INT AS avg_duration_seconds,
    AVG(jsonb_array_length(pages_visited))::NUMERIC(5,1) AS avg_pages_visited,
    MAX(end_time) AS last_session_date
FROM nosql.UserSessions
GROUP BY user_id;


--Витрина: Анализ ценовой истории товаров
DROP MATERIALIZED VIEW nosql.price_analytics;
CREATE MATERIALIZED VIEW nosql.price_analytics AS
SELECT 
    product_id,
    currency,
    COUNT(*) AS price_changes_count,
    MIN(current_price) AS min_price,
    MAX(current_price) AS max_price,
    AVG(current_price)::NUMERIC(10,2) AS avg_price,
    jsonb_array_length(price_changes) AS history_length
FROM nosql.ProductPriceHistory
GROUP BY product_id, currency, jsonb_array_length(price_changes);
